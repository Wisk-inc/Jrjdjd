Running CorX3.8, and how the chat works.

The chat talks to CorX3.8 on your own server. This page has the server code to copy, the four steps to connect it, and how Agent mode’s terminal, files and uploads work.

CorX3.8-27B is too big to run in a browser, so it runs on a GPU you provide — a notebook like molab or Colab, or a rented GPU box. The code below turns that GPU into an OpenAI-compatible API and opens a public URL. You paste that URL into the chat’s settings, and the browser talks to your model directly.

Step 1 · The server

Copy this and run it on a GPU.

Paste the whole thing into one notebook cell and run it. First run installs dependencies and loads the model (1–2 minutes), then it prints a public URL.

"""
================================================================
 CorX3.8-27B — API SERVER for the CorX Labs site
 CorX Labs / Nathan

 Serves the model as an OpenAI-compatible endpoint your website
 can call. transformers backend (vLLM crashed on Qwen3.8 GDN).
================================================================
"""

import os
os.environ["CUDA_VISIBLE_DEVICES"] = "0"
os.environ["PYTORCH_CUDA_ALLOC_CONF"] = "expandable_segments:True"

import subprocess
import sys
import json
import time
import threading
import queue
import uuid
import urllib.request
import gc
import asyncio

# ---------------- SETTINGS ----------------
MODEL_REPO   = "Sigmandndnns/CorX3.8-27B"
HF_TOKEN     = ""
PORT         = 8000
MAX_NEW_TOKENS_CAP = 32000

# Precision. "auto" measures the GPU and picks: full bf16 when the card can
# hold the whole 27B (~54GB) — an RTX PRO 6000 Blackwell, H100, A100 80GB —
# and 4-bit on smaller cards so it still fits in VRAM instead of spilling to
# CPU. Force it with "bf16" or "4bit".
PRECISION = "auto"

# auth
NO_AUTH      = True                     # True = keyless (your "unrestricted")
API_KEY      = "corx-key"               # used only if NO_AUTH = False

# concurrency
QUEUE_TIMEOUT = 120                     # seconds a request waits for the GPU
STREAM_TIMEOUT = 300                    # seconds to wait for a token before
                                        # giving up. Must cover prefill of the
                                        # whole prompt on a 27B model — too low
                                        # and you get "[Generation Error]".
# ------------------------------------------

print("Installing deps (first run only)...")
_deps = ["fastapi", "uvicorn[standard]", "transformers", "accelerate"]
if PRECISION != "bf16":          # may still be needed if auto picks 4-bit
    _deps.append("bitsandbytes")
subprocess.run([sys.executable, "-m", "pip", "install", "-q", "-U", *_deps],
               check=False)

import torch
from transformers import (AutoTokenizer, AutoModelForCausalLM,
                          TextIteratorStreamer)

if HF_TOKEN:
    os.environ["HF_TOKEN"] = HF_TOKEN

print(f"GPUs visible: {torch.cuda.device_count()}")
if torch.cuda.device_count() == 0:
    raise RuntimeError("No GPU attached to this molab session.")

# ---------------- PICK PRECISION FROM THE ACTUAL CARD ----------------
_props = [torch.cuda.get_device_properties(i) for i in range(torch.cuda.device_count())]
TOTAL_VRAM_GB = sum(p.total_memory for p in _props) / 1e9
_cap = f"{_props[0].major}.{_props[0].minor}"
print(f"GPU: {_props[0].name} — {TOTAL_VRAM_GB:.0f} GB total VRAM, compute {_cap}, torch {torch.__version__}")

# A new architecture (Blackwell is sm_120) needs a PyTorch built with kernels
# for it. On an older wheel every CUDA op dies with a bewildering "no kernel
# image is available" — so say plainly what to do before loading 54GB.
try:
    _supported = torch.cuda.get_arch_list()
    if _supported and f"sm_{_props[0].major}{_props[0].minor}" not in _supported:
        print(f"[warn] This torch was built for {_supported} — nothing for sm_{_props[0].major}{_props[0].minor}.")
        print("       If loading fails with 'no kernel image is available', install a matching build:")
        print("       pip install -U --index-url https://download.pytorch.org/whl/cu128 torch")
except Exception:
    pass

# 27B bf16 weights are ~54GB; leave headroom for the KV cache and activations.
_BF16_NEEDS_GB = 62
if PRECISION == "auto":
    LOAD_IN_4BIT = TOTAL_VRAM_GB < _BF16_NEEDS_GB
    print(f"[precision] auto → {'4-bit' if LOAD_IN_4BIT else 'full bf16'} "
          f"({TOTAL_VRAM_GB:.0f} GB {'<' if LOAD_IN_4BIT else '>='} {_BF16_NEEDS_GB} GB)")
else:
    LOAD_IN_4BIT = (PRECISION == "4bit")
    print(f"[precision] forced → {'4-bit' if LOAD_IN_4BIT else 'full bf16'}")
    if not LOAD_IN_4BIT and TOTAL_VRAM_GB < _BF16_NEEDS_GB:
        print(f"[warn] Only {TOTAL_VRAM_GB:.0f} GB of VRAM — bf16 will spill to CPU and be slow.")

# ---------------- LOAD MODEL ----------------
tok_kwargs = {"trust_remote_code": True}
if HF_TOKEN:
    tok_kwargs["token"] = HF_TOKEN

print(f"Loading CorX3.8-27B in {'4-bit' if LOAD_IN_4BIT else 'bf16'} (1-3 min)...")
tok = AutoTokenizer.from_pretrained(MODEL_REPO, **tok_kwargs)
if tok.pad_token is None:
    tok.pad_token = tok.eos_token

load_kwargs = dict(trust_remote_code=True, low_cpu_mem_usage=True)
# transformers renamed torch_dtype -> dtype; pick whichever this version takes
# so the load is clean on both.
import inspect as _inspect
_load_params = _inspect.signature(AutoModelForCausalLM.from_pretrained).parameters
load_kwargs["dtype" if "dtype" in _load_params else "torch_dtype"] = torch.bfloat16
if HF_TOKEN:
    load_kwargs["token"] = HF_TOKEN

# SDPA is PyTorch's built-in fused attention — a solid speedup over the
# eager path on any modern card, and it needs no extra package.
try:
    load_kwargs["attn_implementation"] = "sdpa"
except Exception:
    pass

if LOAD_IN_4BIT:
    # Small card: quantise so the whole model still sits in VRAM. Spilling to
    # CPU is what makes the first token take minutes and trip the timeout.
    from transformers import BitsAndBytesConfig
    load_kwargs["device_map"] = "auto"
    load_kwargs["quantization_config"] = BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_compute_dtype=torch.bfloat16,
        bnb_4bit_quant_type="nf4",
        bnb_4bit_use_double_quant=True)
elif torch.cuda.device_count() == 1:
    # Enough VRAM and one card: put the whole model on GPU 0 explicitly.
    # Pinning it beats device_map="auto", which keeps the offload machinery
    # in play and can still decide to park something on the CPU.
    load_kwargs["device_map"] = {"": 0}
else:
    load_kwargs["device_map"] = "auto"   # shard across the cards

model = AutoModelForCausalLM.from_pretrained(MODEL_REPO, **load_kwargs)
model.eval()
# NOTE: do not set generation_config.cache_implementation = "static" here.
# A static cache sends generate() down the torch.compile / _inductor path,
# which throws inside the worker thread ("Detected that you are using FX
# tracing...") — no tokens are ever produced and the request just hangs.
# The default dynamic cache is correct for this server.
print(f"Model ready. VRAM in use: {torch.cuda.memory_allocated()/1e9:.1f} GB "
      f"of {TOTAL_VRAM_GB:.0f} GB")

# Confirm nothing landed off-GPU — this is the usual root cause of a first
# token that never arrives.
try:
    _dev = set(str(d) for d in model.hf_device_map.values())
    if any(d in ("cpu", "disk") for d in _dev):
        print("[warn] Part of the model is on CPU/disk — generation will be SLOW.")
        print("       Set PRECISION = \"4bit\" so it fits entirely in VRAM.")
    else:
        print(f"[ok] Whole model resident on GPU ({sorted(_dev)}).")
except Exception:
    pass

# ---------------- GENERATION (single GPU, queued) ----------------
_job_lock = threading.Lock()

# Worker wrapper to ensure gradients are off in the background thread.
# If generation dies, end the streamer straight away and hand the reason
# back — otherwise the reader sits waiting for a token that will never come
# and the request only fails much later, on a timeout, with no explanation.
def _generate_thread_worker(kwargs, streamer=None, err_box=None):
    with torch.inference_mode():
        try:
            model.generate(**kwargs)
        except Exception as e:
            import traceback
            traceback.print_exc()
            print(f"\n[Generation Error in thread] {type(e).__name__}: {e}")
            if err_box is not None:
                err_box.append(f"{type(e).__name__}: {e}")
            if streamer is not None:
                try:
                    streamer.end()        # unblock the reader immediately
                except Exception:
                    pass

def _generate(messages, max_new_tokens, temperature, streamer=None, err_box=None):
    prompt = tok.apply_chat_template(messages, tokenize=False,
                                     add_generation_prompt=True)
    inputs = tok(prompt, return_tensors="pt").to(model.device)
    kwargs = dict(**inputs,
                  max_new_tokens=min(int(max_new_tokens), MAX_NEW_TOKENS_CAP),
                  temperature=max(float(temperature), 0.01),
                  top_p=0.9, do_sample=float(temperature) > 0,
                  repetition_penalty=1.05,
                  pad_token_id=tok.pad_token_id or tok.eos_token_id)

    if streamer is not None:
        kwargs["streamer"] = streamer
        # Must target our safe wrapper instead of model.generate directly
        threading.Thread(target=_generate_thread_worker,
                         args=(kwargs, streamer, err_box),
                         daemon=True).start()
        return None

    with torch.inference_mode():
        out = model.generate(**kwargs)
    return tok.decode(out[0][inputs["input_ids"].shape[1]:],
                      skip_special_tokens=True)

# ---------------- OPENAI-COMPATIBLE API + CORS ----------------
from fastapi import FastAPI, Request, HTTPException
from fastapi.responses import StreamingResponse, JSONResponse
from fastapi.middleware.cors import CORSMiddleware
import uvicorn

app = FastAPI(title="CorX3.8 API")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

def check_auth(request: Request):
    if NO_AUTH:
        return
    if request.headers.get("authorization", "") != f"Bearer {API_KEY}":
        raise HTTPException(status_code=401, detail="bad api key")

@app.get("/health")
async def health():
    return {"status": "ok", "model": "corx3.8",
            "vram_gb": round(torch.cuda.memory_allocated()/1e9, 1),
            "busy": _job_lock.locked()}

@app.get("/v1/models")
async def list_models(request: Request):
    check_auth(request)
    return {"object": "list",
            "data": [{"id": "corx3.8", "object": "model",
                      "owned_by": "corx-labs"}]}

@app.post("/v1/chat/completions")
async def chat_completions(request: Request):
    check_auth(request)
    body = await request.json()
    messages = body.get("messages", [])
    if not messages:
        raise HTTPException(status_code=400, detail="messages required")
    max_new = body.get("max_tokens", 2048)
    temperature = body.get("temperature", 0.7)
    stream = bool(body.get("stream", False))
    cid = f"chatcmpl-{uuid.uuid4().hex[:12]}"

    got = _job_lock.acquire(timeout=QUEUE_TIMEOUT)
    if not got:
        raise HTTPException(status_code=503, detail="model busy, retry shortly")

    if not stream:
        try:
            text = _generate(messages, max_new, temperature)
            return JSONResponse({
                "id": cid, "object": "chat.completion", "model": "corx3.8",
                "choices": [{"index": 0, "finish_reason": "stop",
                             "message": {"role": "assistant", "content": text}}],
            })
        except Exception as e:
            import traceback; traceback.print_exc()
            return JSONResponse(status_code=500,
                content={"error": {"message": f"{type(e).__name__}: {e}",
                                    "type": "generation_error"}})
        finally:
            # Bulletproof release & GC
            try: _job_lock.release()
            except RuntimeError: pass
            torch.cuda.empty_cache()
            gc.collect()

    async def event_stream():
        try:
            # This is the wait for the NEXT token, and the first one only
            # arrives after the whole prompt is prefilled. A 27B model with a
            # long system prompt (and any CPU offload) can take well over a
            # minute to get there, so a short timeout here shows up in the
            # chat as "[Generation Error]" even though nothing actually failed.
            streamer = TextIteratorStreamer(
                tok, skip_prompt=True, skip_special_tokens=True, timeout=STREAM_TIMEOUT)
            err_box = []          # filled by the worker if generate() throws
            _generate(messages, max_new, temperature, streamer=streamer,
                      err_box=err_box)

            _DONE = object() # Sentinel value to prevent StopIteration Exception
            sent_any = False

            while True:
                try:
                    # Pass _DONE as the default value to next() so it doesn't throw an error to asyncio
                    chunk = await asyncio.to_thread(next, streamer, _DONE)

                    if chunk is _DONE:
                        break # Stream finished successfully

                    data = {"id": cid, "object": "chat.completion.chunk",
                            "model": "corx3.8",
                            "choices": [{"index": 0, "delta": {"content": chunk},
                                         "finish_reason": None}]}
                    sent_any = True
                    yield f"data: {json.dumps(data)}\n\n"

                except queue.Empty:
                    print(f"[api] No token for {STREAM_TIMEOUT}s — giving up. "
                          f"Raise STREAM_TIMEOUT, shorten the prompt, or load in 4-bit.")
                    err = {"id": cid, "object": "chat.completion.chunk",
                           "model": "corx3.8",
                           "choices": [{"index": 0, "delta": {"content":
                                        f"\n[No token produced within {STREAM_TIMEOUT}s. "
                                        f"The model is likely still prefilling — raise STREAM_TIMEOUT "
                                        f"in the server, or load the model in 4-bit so it fits the GPU.]"},
                                        "finish_reason": "error"}]}
                    yield f"data: {json.dumps(err)}\n\n"
                    break

            # generate() died before producing anything: say what actually
            # went wrong instead of returning an empty, silent reply.
            if err_box and not sent_any:
                err = {"id": cid, "object": "chat.completion.chunk",
                       "model": "corx3.8",
                       "choices": [{"index": 0, "delta": {"content":
                                    f"\n[Generation failed on the server: {err_box[0]}]"},
                                    "finish_reason": "error"}]}
                yield f"data: {json.dumps(err)}\n\n"

            done = {"id": cid, "object": "chat.completion.chunk",
                    "model": "corx3.8",
                    "choices": [{"index": 0, "delta": {},
                                 "finish_reason": "stop"}]}
            yield f"data: {json.dumps(done)}\n\n"
            yield "data: [DONE]\n\n"
        finally:
            # Bulletproof release & GC for streaming
            try: _job_lock.release()
            except RuntimeError: pass
            torch.cuda.empty_cache()
            gc.collect()

    return StreamingResponse(event_stream(), media_type="text/event-stream")

# ---------------- CLOUDFLARE QUICK TUNNEL (zero account) ----------------
import re
import platform

def get_cloudflared():
    path = os.path.abspath("./cloudflared")
    if os.path.exists(path):
        return path
    arch = platform.machine().lower()
    if arch in ("x86_64", "amd64"):
        url = ("https://github.com/cloudflare/cloudflared/releases/latest/"
               "download/cloudflared-linux-amd64")
    elif "aarch64" in arch or "arm64" in arch:
        url = ("https://github.com/cloudflare/cloudflared/releases/latest/"
               "download/cloudflared-linux-arm64")
    else:
        url = ("https://github.com/cloudflare/cloudflared/releases/latest/"
               "download/cloudflared-linux-amd64")
    print(f"[tunnel] downloading cloudflared ({arch})...")
    urllib.request.urlretrieve(url, path)
    os.chmod(path, 0o755)
    return path

def start_cloudflare_tunnel(port, timeout=40):
    binary = get_cloudflared()
    proc = subprocess.Popen(
        [binary, "tunnel", "--no-autoupdate", "--url", f"http://localhost:{port}"],
        stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, bufsize=1)
    url = None
    url_re = re.compile(r"https://[-a-z0-9]+\.trycloudflare\.com")
    start = time.time()
    while time.time() - start < timeout:
        line = proc.stdout.readline()
        if not line:
            if proc.poll() is not None:
                break
            continue
        m = url_re.search(line)
        if m:
            url = m.group(0)
            break
    def _drain():
        for _ in proc.stdout:
            pass
    threading.Thread(target=_drain, daemon=True).start()
    return url, proc

Which GPU? CorX3.8 needs roughly 24 GB of GPU memory in 4-bit, or about 54 GB in full bf16. You don't have to choose: PRECISION = "auto" measures the card at start-up and loads full bf16 when there is room for it (RTX PRO 6000 Blackwell, H100, A100 80GB), dropping to 4-bit only on smaller cards. Force it either way with PRECISION = "bf16" or "4bit".

On a single card that fits the model, it is pinned to GPU 0 rather than left to device_map="auto" — that keeps the offload machinery out of the picture entirely. The cell prints which precision it chose and confirms the whole model is resident on the GPU; if anything lands on CPU or disk it says so, because that is what makes the first token take minutes.

On a brand-new card (Blackwell is sm_120) PyTorch must have kernels built for it, or every CUDA call fails with “no kernel image is available”. The script checks and, if your build is too old, prints the install line: pip install -U --index-url https://download.pytorch.org/whl/cu128 torch.

Step 2 · Connect

Paste the URL into the chat.

  1. Wait for “API IS LIVE”

    When the cell prints CorX3.8-27B API IS LIVE, it shows a Public URL like https://something.trycloudflare.com/v1. Copy it without the /v1 — the chat adds that for you.

  2. Open the chat and its settings

    Go to the chat and click the model name at the top (or the gear icon) to open Model & connection. With CorX3.8 selected, paste the URL into Server address and press Save.

  3. Watch the status dot

    The dot by the title turns green · Online when it connects (it checks the server’s /health). Red means it can’t reach the server — check the cell is still running and the URL is current.

  4. Leave the cell running

    The server lives as long as that cell runs. The code blocks on purpose so the notebook doesn’t tear it down, and restarts the tunnel if it drops.

Or bring your own model. The same Model & connection panel has a provider picker: pick OpenRouter, Claude, DeepSeek or OpenAI, paste that provider’s API key (stored in your browser only, sent only to that provider), and choose a model. The whole agent — search, terminal, files, plan — then runs on that model instead. Two of these call the provider straight from the browser and just work: OpenRouter (one key, most models) and Claude. DeepSeek and OpenAI block direct browser calls, so they’re marked proxy and will hit a CORS wall unless you front them yourself — use OpenRouter to reach GPT or DeepSeek models from the browser.

Step 3 · Agent tools

It runs code, files and a plan on its own.

There is no toggle to switch on — every conversation can use tools: run Python, make and edit files, install packages and plan a task. CorX3.8 decides for itself when a tool would help and calls it without asking permission first. The tools run in a real Python sandbox inside your browser tab — not on the model server — so you can watch every command in the Terminal panel and nothing touches your computer outside the tab.

What it can do

  • Run Python and see the real output in the Terminal
  • Create, edit, save and delete files in the Files panel
  • Install Python packages with micropip
  • Publish a plan and tick off each step as it works
  • Take a zip or file you upload and work on it, with Python's own zipfile module
  • Search the web and fetch a page's real content, on its own initiative
  • Recall relevant bits of your other saved chats
  • Pull real pictures into the chat with find_image (Wikimedia Commons, credited)
  • Read its own traceback when code fails and try to fix it on the next round

The panel, search & uploads

Open the work panel with the panel button by the title. It has three tabs: Terminal (commands as they run, and you can type Python yourself), Files (everything the sandbox holds, shared across every chat in this tab — click one to edit, save or delete), and Plan (the live checklist).

A real web search runs through /api/search (DuckDuckGo, no key) with a sliding row of site logos while it searches, then a Searched the web dropdown you can open to see every link and snippet. /api/fetch reads a page's actual content, including any code shown on it.

The upload button in the composer stages files — including .zip archives — into the sandbox at /work. Ask the agent to unzip, read or transform them.

Honest note on tools. The model server is a plain chat endpoint with no built-in tool-calling, so every system prompt teaches CorX3.8 a small text format for asking to run a tool. Whether it uses tools well depends on the model following that format — it is a Patois-focused fine-tune, not a dedicated tool-use model, so expect it to be better at conversation than at long autonomous chains. If it ever ignores a tool it should have used, ask it directly and it will call it.

Step 4 · Effort

Low to Max — a real dial, not a label.

The Effort selector in the composer changes actual request parameters: how many tokens the model is allowed per reply, how many tool-call rounds the agent gets before it must stop, how many search results it pulls back, and how hard the system prompt tells it to think, weigh alternatives, verify its own steps and search when unsure. Higher effort genuinely takes longer and spends more tokens — that trade is the point, for a task that is actually hard.

Extra and Max also add a second pass: after producing code or a solution, the model is told to stop and deliberately review its own work for bugs or missed cases — as if critiquing someone else's pull request — before presenting it as done, and to say so plainly and correct itself if it realises mid-answer that something it already said was wrong.

Effort is set per conversation and remembered when you come back to that chat.

Conversations & profile

Saved in this browser, nowhere else.

Conversations

Every chat lives in the sidebar — switch between them, delete one, or start a New chat. Messages, the plan, the agent toggle and the effort level are all saved per conversation and restored exactly as you left them after a refresh, a closed tab, or days away. The Python sandbox itself is not per-conversation — it is one real Python process for the whole tab, so its files are shared across every chat you have open.

Interrupted runs resume themselves

If you refresh mid-task, the chat notices on reload and starts picking the run back up automatically after a few seconds — Dismiss if you'd rather not. Because the Python sandbox resets on every reload, the agent is told plainly that it did, and to check what actually exists with list_files before assuming anything survived.

Profile

Click your name at the bottom of the sidebar to set a name and a photo. Both are saved locally and used to label your messages — never sent anywhere, and not part of what the model itself receives.

Cross-chat memory

With Memory on in your profile, the agent can call search_memory to grep your other saved conversations for relevant context — a real keyword search over what is in this browser, not a hidden server-side record.

Honest limits

What this set-up is, and isn’t.

  • The URL is ephemeral. A Cloudflare quick tunnel gives a new address every restart, and the notebook recycles. When the chat goes Offline, re-run the cell and paste the new URL. A permanent public chat needs a persistent host and a fixed domain.

def ensure_packages(pkgs):

        try:
            __import__(mod)

        ["uv", "pip", "install", "-q", "--python", sys.executable] + missing,

def free_port(port):
    killed = []
    for cmd in (["fuser", "-k", f"{port}/tcp"], ["lsof", "-ti", f":{port}"]):
        try:
            if cmd[0] == "fuser":
                sh(cmd, check=False, stdout=subprocess.DEVNULL,
                   stderr=subprocess.DEVNULL)
            else:
                out = subprocess.check_output(cmd, text=True,
                                              stderr=subprocess.DEVNULL).split()
                for pid in out:
                    os.kill(int(pid), signal.SIGKILL)
                    killed.append(pid)
        except Exception:
            continue
    if killed:
        print(f"[env] freed port {port} (killed {', '.join(killed)})")
        time.sleep(1)

def _works(path):
    """Does this binary actually run here?

    The old version only caught exceptions, so a binary that started and died
    on a missing .so returned True. Check the exit code, and surface the last
    line of output so a failure is diagnosable rather than mysterious."""
    if not (os.path.isfile(path) and os.access(path, os.X_OK)):
        return False
    try:
        r = subprocess.run([path, "--version"], capture_output=True, text=True,
                           timeout=120,
                           env=dict(os.environ, LD_LIBRARY_PATH=os.path.dirname(path)))
    except Exception as e:
        print(f"[build]   won't start: {type(e).__name__}: {e}")
        return False
    if r.returncode != 0:
        out = ((r.stderr or "") + (r.stdout or "")).strip().splitlines()
        print(f"[build]   exited {r.returncode}: "
              f"{out[-1][:200] if out else '(no output)'}")
        return False
    return True


# A prebuilt binary built on Ubuntu often refuses to start on this container
# for one dull reason: a shared library it links against is absent or named
# differently here. That returns _works() == False and the whole prebuilt path
# is abandoned — which is very likely why the Vulkan binary was skipped and the
# run fell through to CUDA. So read ldd, install what is actually missing, and
# retry, instead of giving up on a fixable link error.

    "libgomp.so": ["libgomp1"],
    "libcurl.so": ["libcurl4t64", "libcurl4"],
    "libstdc++.so": ["libstdc++6"],

    "libgfortran.so": ["libgfortran5"],

def _missing_libs(path):
    try:
        r = subprocess.run(["ldd", path], capture_output=True, text=True,
                           env=dict(os.environ, LD_LIBRARY_PATH=os.path.dirname(path)))
    except Exception:
        return []
    miss = []
    for line in (r.stdout + r.stderr).splitlines():
        m = re.search(r"(\S+)\s*=>\s*not found", line)
        if m:
            miss.append(m.group(1))
    return miss


def ensure_runnable(path):
    """_works(), and if it fails on a missing .so, install it and try once more."""
    if _works(path):
        return True
    miss = _missing_libs(path)
    if not miss:
        return False
    print(f"[build]   binary needs libraries this box lacks: {', '.join(miss)}")
    for so in miss:
        stem = so.split(".so")[0] + ".so"
        for pkg in _SONAME_PKG.get(stem, []):
            if apt_has(pkg) and apt([pkg]):
                break
    still = _missing_libs(path)
    if still:
        print(f"[build]   still missing after install: {', '.join(still)} "
              "(no apt package maps to these).")
    return _works(path)


def _install_binary(src_dir):

    Also pulls a sibling lib/ if the archive uses that layout."""
    os.makedirs(BIN_DIR, exist_ok=True)
    dirs = [src_dir]
    sib = os.path.join(os.path.dirname(src_dir), "lib")
    if os.path.isdir(sib):
        dirs.append(sib)
    for d in dirs:
        for f in os.listdir(d):
            s = os.path.join(d, f)
            if os.path.isfile(s):

def distro_candidates():
    """apt repo names to try, in order.

    NVIDIA publishes no repo for Debian 13 (trixie). Your log shows debian13
    accepting the keyring and then failing to locate a single package: apt-get
    update was 404ing against a directory that does not exist. Map trixie onto
    debian12 and keep the Ubuntu repos behind it."""
    try:
        osr = dict(l.strip().split("=", 1) for l in open("/etc/os-release") if "=" in l)
        did = osr.get("ID", "").strip('"')
        ver = osr.get("VERSION_ID", "").strip('"').replace(".", "")
    except Exception:
        did, ver = "", ""
    alias = {"debian13": "debian12", "debian14": "debian12",
             "ubuntu2410": "ubuntu2404", "ubuntu2504": "ubuntu2404",
             "ubuntu2510": "ubuntu2404", "ubuntu2604": "ubuntu2404"}
    native = f"{did}{ver}"
    out = []
    for d in (alias.get(native, native), "debian12", "ubuntu2404", "ubuntu2204"):
        if d and d not in out:
            out.append(d)
    return out


def apt_has(pkg):
    """Is this package actually installable right now? apt-cache policy prints
    'Candidate: (none)' when it is not, which beats guessing from log text."""
    try:
        out = subprocess.check_output(["apt-cache", "policy", pkg], text=True,
                                      stderr=subprocess.DEVNULL)
    except Exception:
        return False
    m = re.search(r"Candidate:\s*(\S+)", out)
    return bool(m and m.group(1) != "(none)")


def drop_cuda_sources():

def trust_cuda_sources():
    """Mark the NVIDIA CUDA apt source trusted.

    This is the fix for "OpenPGP signature verification failed" on this box.
    NVIDIA publishes no Debian 13 (trixie) repo, so we borrow debian12; but
    Debian 13's apt (3.0) then rejects the InRelease signature and every
    candidate looks unreachable. The keyring is installed and the repo is
    NVIDIA's own host over HTTPS, so tell apt to trust it rather than fail on a
    cross-release signature check. Handles both the one-line (.list) and the
    deb822 (.sources) formats a keyring might drop."""
    touched = False
    for f in glob.glob("/etc/apt/sources.list.d/cuda*"):
        try:
            txt = open(f).read()
        except OSError:
            continue
        if f.endswith(".sources"):
            # deb822: add "Trusted: yes" to each stanza (blank-line separated)
            # that does not already carry it.
            stanzas = re.split(r"\n\s*\n", txt.strip())
            fixed = []
            for st in stanzas:
                if st.strip() and not re.search(r"(?im)^\s*trusted:", st):
                    st = st.rstrip() + "\nTrusted: yes"
                    touched = True
                fixed.append(st)
            new = "\n\n".join(fixed) + "\n"
        else:
            new_lines = []
            for line in txt.splitlines():
                s = line.strip()
                if s.startswith("deb") and "trusted=yes" not in s:
                    if "[" in s:
                        s = s.replace("[", "[trusted=yes ", 1)
                    else:
                        s = s.replace("deb ", "deb [trusted=yes] ", 1)
                    touched = True
                new_lines.append(s)
            new = "\n".join(new_lines) + "\n"
        try:
            open(f, "w").write(new)
        except OSError:
            pass
    return touched


def install_cuda_toolkit():
    """Install just the pieces llama.cpp needs: nvcc, cudart headers, cuBLAS
    dev. About a third of the full toolkit. Needs root."""
    if not IS_ROOT:
        print("[cuda] no toolkit and not root, so it cannot be installed here.")
        return None

    apt(["ca-certificates", "curl", "gnupg"])
    try:
        arch_i = int(CUDA_ARCH)
    except ValueError:
        arch_i = 0
    # sm_120 (Blackwell, incl. RTX PRO 6000) needs 12.8 or newer. Older cards
    # can fall further back.
    versions = (["13-0", "12-9", "12-8"] if arch_i >= 120
                else ["13-0", "12-9", "12-8", "12-6", "12-4"])

    for d in distro_candidates():
        url = ("https://developer.download.nvidia.com/compute/cuda/repos/"
               f"{d}/x86_64/cuda-keyring_1.1-1_all.deb")
        print(f"[cuda] trying NVIDIA's apt repo for {d} ...")
        deb = "/tmp/cuda-keyring.deb"
        try:
            urllib.request.urlretrieve(url, deb)
        except Exception as e:
            print(f"[cuda]   no keyring for {d} ({type(e).__name__})")
            continue
        if sh(["dpkg", "-i", deb], check=False, stdout=subprocess.DEVNULL,
              stderr=subprocess.DEVNULL).returncode != 0:
            print(f"[cuda]   keyring would not install for {d}")
            continue

        # Trust the repo BEFORE updating: on Debian 13 the InRelease signature
        # is rejected ("OpenPGP signature verification failed"), so without this
        # the update below fails and the candidate is discarded.
        trust_cuda_sources()

        # build 9 tested `"404" in blob or "Err:" in blob` against the WHOLE
        # apt-get update output. The NVIDIA domain appears in normal successful
        # Get: lines, so any unrelated broken source anywhere in sources.list
        # made every candidate look dead — which is why debian12, ubuntu2404
        # and ubuntu2204 were all rejected despite all three existing. Match
        # per line, and only on lines that are actually errors for this repo.
        upd = sh("apt-get update -o Acquire::Retries=2", check=False,
                 capture_output=True, text=True)
        lines = ((upd.stdout or "") + (upd.stderr or "")).splitlines()
        # A signature warning is no longer fatal now the repo is trusted; only a
        # genuine fetch failure (404/could-not-connect) means "not reachable".
        broken = [l for l in lines
                  if l.strip().startswith(("Err:", "E:"))
                  and "developer.download.nvidia.com" in l
                  and "signature" not in l.lower()
                  and "signed" not in l.lower()]
        if broken:
            print(f"[cuda]   {d} repo is not reachable:")
            for l in broken[:3]:
                print("           " + l.strip()[:150])
            drop_cuda_sources()
            continue
        _APT_UPDATED[0] = True

        # Ask apt what it can actually see, rather than inferring it from log
        # text. "Candidate: (none)" is unambiguous; parsing update output is not.
        for ver in versions:
            if not apt_has(f"cuda-nvcc-{ver}"):
                continue
            pkgs = [f"cuda-nvcc-{ver}", f"cuda-cudart-dev-{ver}",
                    f"libcublas-dev-{ver}", f"cuda-cccl-{ver}"]
            pkgs = [p for p in pkgs if apt_has(p)]
            print(f"[cuda] installing {' '.join(pkgs)} (about 1 GB, once)...")
            if apt(pkgs):
                nv = find_nvcc()
                if nv:
                    print(f"[cuda] toolkit installed: {nv}")
                    return nv
        # Fall through to the NEXT candidate, rather than break-ing out of the
        # loop the way build 8 did.
        print(f"[cuda]   {d} is reachable but ships none of {', '.join(versions)}.")
        drop_cuda_sources()

    print("[cuda] could not install a CUDA toolkit from NVIDIA's apt repo.")
    return None

def clone_llama():
    """Get a llama.cpp source tree, and be certain it is actually a source tree.

    This is what broke build 10. The check was `if not os.path.isdir(LLAMA_SRC)`,
    which is not the same question. build_vulkan() calls this first; when its
    clone failed it left an empty /marimo/llama.cpp behind, and the CUDA build
    that followed saw a directory, skipped the clone, and handed cmake a folder
    with nothing in it — "does not appear to contain CMakeLists.txt". Test for
    the file that proves the checkout, not for the directory."""
    apt(["build-essential", "cmake", "git", "ccache", "pkg-config"])
    if not shutil.which("cmake"):
        raise RuntimeError("cmake is not installed and could not be installed "
                           "(root=%s). Nothing can be compiled here." % IS_ROOT)
    if not shutil.which("git"):
        raise RuntimeError("git is not installed and could not be installed "
                           "(root=%s)." % IS_ROOT)

    marker = os.path.join(LLAMA_SRC, "CMakeLists.txt")
    if os.path.isfile(marker):
        return

    tail = []
    for attempt in range(3):
        if os.path.exists(LLAMA_SRC):
            print("[build] llama.cpp checkout is incomplete — removing and re-cloning.")
            shutil.rmtree(LLAMA_SRC, ignore_errors=True)
        print(f"[build] cloning llama.cpp (attempt {attempt + 1}/3)...")
        r = sh(["git", "clone", "--depth", "1",
                "https://github.com/ggml-org/llama.cpp", LLAMA_SRC],
               check=False, capture_output=True, text=True)
        if os.path.isfile(marker):
            return
        tail = ((r.stdout or "") + (r.stderr or "")).strip().splitlines()[-4:]
        for l in tail:
            print("        " + l[:160])
        time.sleep(2 * (attempt + 1))

    # Leave nothing behind: a half-cloned directory here is exactly what made
    # the *next* build skip cloning and fail on a missing CMakeLists.txt.
    shutil.rmtree(LLAMA_SRC, ignore_errors=True)
    raise RuntimeError(
        "Could not clone llama.cpp after 3 attempts. Last output: "
        + (" | ".join(tail) if tail else "(none)"))


def _run_capture(cmd):
    """Run and hand back (returncode, combined output). Never hide the reason a
    build died — build 9 raised a bare CalledProcessError whose message was just
    the argv, so the real cause was only visible by scrolling the cell."""
    r = subprocess.run(cmd, capture_output=True, text=True)
    return r.returncode, (r.stdout or "") + (r.stderr or "")


def _print_tail(what, rc, out, n=30):
    print(f"[build] {what} failed (exit {rc}):")
    for l in out.strip().splitlines()[-n:]:
        print("        " + l[:160])


def cmake_build(build_dir, flags):
    base = ["-DLLAMA_CURL=OFF", "-DCMAKE_BUILD_TYPE=Release"]
    rc, out = _run_capture(["cmake", "-S", LLAMA_SRC, "-B", build_dir] + flags + base)
    if rc != 0:
        # A CMakeCache.txt from an earlier, different configure (different source
        # path or compiler) poisons every later one, and the message for that is
        # not obviously about the cache. Wipe and try once more before believing
        # the failure is real.
        print("[build] configure failed — clearing the build directory and retrying.")
        shutil.rmtree(build_dir, ignore_errors=True)
        rc, out = _run_capture(["cmake", "-S", LLAMA_SRC, "-B", build_dir] + flags + base)
    if rc != 0:
        _print_tail("cmake configure", rc, out)
        raise RuntimeError(f"cmake configure failed: {_short_reason(out)}")

    rc, out = _run_capture(["cmake", "--build", build_dir, "--config", "Release",
                            "-j", str(os.cpu_count() or 4), "--target", "llama-server"])
    if rc != 0:
        _print_tail("cmake build", rc, out)
        raise RuntimeError(f"cmake build failed: {_short_reason(out)}")

    built = os.path.join(build_dir, "bin", "llama-server")
    if not os.path.exists(built):
        raise RuntimeError("Build finished but llama-server is missing — see the log.")
    if not _install_binary(os.path.dirname(built)):
        raise RuntimeError("Built llama-server will not run here.")
    return built


def _short_reason(out):
    """The one line worth putting in an exception message."""
    for l in out.strip().splitlines():
        s = l.strip()
        if s.startswith(("CMake Error", "error:", "fatal error:")) or "unsupported" in s:
            return s[:200]
    tail = out.strip().splitlines()
    return tail[-1][:200] if tail else "no output"


def _older_gcc():
    """A host compiler nvcc will accept, if the default one is too new.

    CUDA pins the maximum supported GCC; Debian 13 ships a newer one than some
    toolkits allow, and nvcc then stops with "unsupported GNU version"."""
    for v in ("14", "13", "12", "11"):
        exe = shutil.which(f"g++-{v}")
        if exe:
            return exe
        if apt_has(f"g++-{v}") and apt([f"g++-{v}"]):
            exe = shutil.which(f"g++-{v}")
            if exe:
                return exe
    return None


def build_cuda(nvcc):
    ver = nvcc_version(nvcc)
    if ver:
        print(f"[build] CUDA toolkit {ver[0]}.{ver[1]} at {nvcc}")
        if int(CUDA_ARCH) >= 120 and ver < (12, 8):
            print(f"[warn] sm_{CUDA_ARCH} needs CUDA 12.8 or newer and this is "
                  f"{ver[0]}.{ver[1]}. The build will probably fail.")
    clone_llama()
    toolkit_root = os.path.dirname(os.path.dirname(nvcc))
    flags = ["-DGGML_CUDA=ON",
             f"-DCMAKE_CUDA_ARCHITECTURES={CUDA_ARCH}",
             f"-DCMAKE_CUDA_COMPILER={nvcc}",
             f"-DCUDAToolkit_ROOT={toolkit_root}"]
    print(f"[build] compiling for sm_{CUDA_ARCH} (5-15 min, cached afterwards)...")
    try:
        cmake_build(os.path.join(LLAMA_SRC, "build-cuda"), flags)
    except RuntimeError as e:
        # The one CUDA failure worth a second attempt: a host compiler newer than
        # this toolkit accepts. Everything else is reported as-is.
        if "unsupported GNU version" not in str(e) and "unsupported" not in str(e).lower():
            raise
        gxx = _older_gcc()
        if not gxx:
            raise
        print(f"[build] host compiler rejected by nvcc — retrying with {gxx}")
        cmake_build(os.path.join(LLAMA_SRC, "build-cuda"),
                    flags + [f"-DCMAKE_CUDA_HOST_COMPILER={gxx}"])
    print(f"[build] ok: {SERVER_BIN}")
    return "cuda"


def have_spirv_headers():
    """ggml-vulkan includes <spirv/unified1/spirv.hpp>. glslc alone does not
    supply it — build 9 installed glslc, configured, and died on SPIRV-Headers."""
    for d in ("/usr/include/spirv/unified1", "/usr/local/include/spirv/unified1"):
        if glob.glob(os.path.join(d, "spirv.*")):
            return True
    return False


def vendor_spirv_headers():
    """Fallback when the distro has no spirv-headers package: take them from
    Khronos directly. Headers only, no build, a few hundred KB."""
    dst = "/usr/local/include/spirv"
    try:
        tmp = "/tmp/SPIRV-Headers"
        shutil.rmtree(tmp, ignore_errors=True)
        sh(["git", "clone", "--depth", "1",
            "https://github.com/KhronosGroup/SPIRV-Headers", tmp], check=True,
           stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        src = os.path.join(tmp, "include", "spirv")
        if not os.path.isdir(src):
            return False
        shutil.rmtree(dst, ignore_errors=True)
        shutil.copytree(src, dst)
        print(f"[build] vendored SPIRV-Headers into {dst}")
        return True
    except Exception as e:
        print(f"[build] could not vendor SPIRV-Headers ({type(e).__name__}: {e})")
        return False


def build_vulkan():
    """Compile with Vulkan. The cheap path when the NVIDIA driver is present
    but the toolkit is not: a shader toolchain instead of 1 GB of nvcc.

    Blackwell exposes NV_coopmat2 through Vulkan, so this is a fast path on an
    RTX PRO 6000, not a consolation prize."""
    try:
        clone_llama()
    except Exception as e:
        print(f"[build] cannot compile ({e})")
        return None

    # One at a time: names vary across distros and one bad name must not take
    # the whole batch down with it.
    for p in ("libvulkan-dev", "vulkan-headers", "glslc", "libshaderc-dev",
              "glslang-tools", "glslang-dev", "libglslang-dev",
              "spirv-headers", "spirv-tools", "libspirv-tools-dev"):
        apt([p])

    if not shutil.which("glslc"):
        apt(["shaderc"])
    if not shutil.which("glslc"):
        print("[build] glslc is unavailable, so the Vulkan shaders cannot be "
              "compiled. Skipping the Vulkan source build.")
        return None
    if not have_spirv_headers() and not vendor_spirv_headers():
        print("[build] no SPIR-V headers available; skipping the Vulkan build.")
        return None

    print("[build] compiling llama.cpp with Vulkan (5-12 min, cached afterwards)...")
    try:
        cmake_build(os.path.join(LLAMA_SRC, "build-vulkan"),
                    ["-DGGML_VULKAN=ON",
                     "-DCMAKE_CXX_FLAGS=-I/usr/local/include"])
    except Exception as e:
        print(f"[build] Vulkan build failed ({type(e).__name__}) — see the lines "
              "above for the actual cmake error.")
        return None
    print(f"[build] ok: {SERVER_BIN}")
    return "vulkan"

def vulkan_device_kind():
    """'nvidia', 'software', 'other' or None, read from the ICD manifests
    rather than by running vulkaninfo, which is usually not installed. The
    absence of this check is what made an earlier build look healthy while it
    ran on the CPU."""
    icds = []
    for d in ("/usr/share/vulkan/icd.d", "/etc/vulkan/icd.d",
              "/usr/local/share/vulkan/icd.d"):
        icds += glob.glob(os.path.join(d, "*.json"))
    names = " ".join(os.path.basename(i).lower() for i in icds)
    if not icds:
        return None
    if "nvidia" in names:
        return "nvidia"
    if "lvp" in names or "lavapipe" in names or "swrast" in names:
        return "software"
    return "other"


def try_prebuilt(kind):
    """Fetch a published llama.cpp binary. kind is 'cuda' or 'vulkan'.

    Tries EVERY matching asset, and prints what was on offer when none match.
    The old version returned None from inside the loop, so one bad asset ended
    the search and the log never said why."""
    if platform.machine().lower() not in ("x86_64", "amd64"):
        return None
    url = "https://api.github.com/repos/ggml-org/llama.cpp/releases/latest"
    req = urllib.request.Request(url, headers={"User-Agent": "corx-heretic"})
    try:
        with urllib.request.urlopen(req, timeout=30) as r:
            rel = json.load(r)
    except Exception as e:
        print(f"[build] could not read release info ({type(e).__name__}: {e})")
        return None

    all_names = [a["name"] for a in rel.get("assets", [])]
    matches = []
    for a in rel.get("assets", []):
        n = a["name"].lower()
        if (("ubuntu" in n or "linux" in n) and kind in n
                and ("x64" in n or "amd64" in n)
                and (n.endswith(".tar.gz") or n.endswith(".zip"))):
            matches.append(a)

    if not matches:
        print(f"[build] no {kind} binary published for Linux x64. Assets in "
              f"{rel.get('tag_name', '?')}: {', '.join(all_names[:14])}")
        return None

    install_runtime_libs()
    for a in matches:
        print(f"[build] downloading prebuilt {kind} binary ({a['name']})...")
        try:
            dl = os.path.join("/tmp", a["name"])
            urllib.request.urlretrieve(a["browser_download_url"], dl)
            ex = f"/tmp/llama-{kind}-extract"
            shutil.rmtree(ex, ignore_errors=True)
            os.makedirs(ex, exist_ok=True)
            if dl.endswith(".zip"):
                with zipfile.ZipFile(dl) as zf:
                    zf.extractall(ex)
            else:
                with tarfile.open(dl, "r:gz") as tf:
                    tf.extractall(ex)
            found = glob.glob(os.path.join(ex, "**", "llama-server"), recursive=True)
            if not found:
                print("[build]   archive contains no llama-server.")
                continue
            if _install_binary(os.path.dirname(found[0])):
                print(f"[build] using prebuilt {kind} binary.")
                return kind
            print("[build]   that archive would not run here; trying the next.")
        except Exception as e:
            print(f"[build]   fetch failed ({type(e).__name__}: {e})")
    return None

def acquire_backend():
    if BACKEND in ("cuda", "vulkan"):
        print(f"[build] BACKEND is pinned to {BACKEND}.")
    want_cuda = BACKEND in ("auto", "cuda")
    want_vulkan = BACKEND in ("auto", "vulkan")
    failures = []

    def attempt(label, fn):
        """A rung that fails must not kill the run — the next one may work.
        Only an exhausted ladder is fatal, and then we report every reason."""
        try:
            return fn()
        except Exception as e:
            print(f"[build] {label} failed: {e}")
            failures.append(f"{label}: {e}")
            return None

    # 1. Toolkit already present: compile CUDA, the fastest thing to run.
    if want_cuda:
        nvcc = find_nvcc()
        if nvcc:
            got = attempt("CUDA build", lambda: build_cuda(nvcc))
            if got:
                return got
        else:
            print("[build] no nvcc here — the driver is installed, the CUDA "
                  "toolkit is not.")
        got = attempt("prebuilt CUDA", lambda: try_prebuilt("cuda"))
        if got:
            return got

    # 2. Vulkan, but only against the real driver.
    if want_vulkan:
        kind = vulkan_device_kind()
        if kind is None:
            print("[build] no Vulkan driver installed; installing libvulkan1 "
                  "(NOT mesa, which is the software one)...")
            apt(["libvulkan1"])
            kind = vulkan_device_kind()
        if kind == "nvidia":
            print("[build] real NVIDIA Vulkan driver found — a genuine GPU path, "
                  "somewhat slower than CUDA.")
            got = attempt("prebuilt Vulkan", lambda: try_prebuilt("vulkan"))
            if got:
                return got
            got = attempt("Vulkan build", build_vulkan)
            if got:
                return got
        elif kind == "software":
            print("[build] the only Vulkan driver here is a software one "
                  "(lavapipe). Refusing it: that runs the model on the CPU while "
                  "reporting layers offloaded to GPU.")
        elif kind is not None:
            print(f"[build] Vulkan drivers present but not NVIDIA ({kind}); skipping.")

    # 3. Nothing free left. Pay for a toolkit and compile.
    if want_cuda and INSTALL_CUDA_TOOLKIT and not find_nvcc():
        print("[cuda] no GPU backend yet — installing a CUDA toolkit.")
        nvcc = install_cuda_toolkit()
        if nvcc:
            got = attempt("CUDA build (fresh toolkit)", lambda: build_cuda(nvcc))
            if got:
                return got

    raise RuntimeError(
        "No GPU backend could be set up.\n"
        f"  - nvcc: {find_nvcc() or 'not found'}\n"
        f"  - Vulkan ICD: {vulkan_device_kind() or 'no driver installed'}\n"
        f"  - root: {IS_ROOT}\n"
        + ("What was tried:\n" + "".join(f"  - {f}\n" for f in failures) if failures else "")
        + "Fix one of these and re-run:\n"
        "  * install glslc + libvulkan-dev so the Vulkan build can compile, or\n"
        "  * use a runtime image that ships the CUDA toolkit (nvcc), or\n"
        "  * run as root so this cell can install one.\n"
        "Do not work around this by letting it run on the CPU — a 27B there is "
        "minutes per reply.")

_SPLIT_RE = re.compile(r"^(?P<stem>.+)-(?P<idx>\d{5})-of-(?P<tot>\d{5})\.gguf$", re.I)


def repo_files_with_sizes():
    """?blobs=true so each sibling carries a byte size — needed to check the
    disk before committing to a multi-gigabyte download."""
    url = f"https://huggingface.co/api/models/{MODEL_REPO}?blobs=true"
    req = urllib.request.Request(url, headers={"User-Agent": "corx-heretic"})
    if HF_TOKEN:
        req.add_header("Authorization", f"Bearer {HF_TOKEN}")
    try:
        with urllib.request.urlopen(req, timeout=30) as r:
            sibs = json.load(r).get("siblings", [])
    except urllib.error.HTTPError as e:
        if e.code in (401, 403):
            raise RuntimeError(
                f"{MODEL_REPO} is gated or private ({e.code}). Accept the terms on "
                "the model page and set HF_TOKEN at the top of this cell.") from e
        if e.code == 404:
            raise RuntimeError(
                f"{MODEL_REPO} does not exist on the Hub (404). Check the exact "
                "owner/name — it is case-sensitive and must be a repo that "
                "actually ships .gguf files.") from e
        raise
    return {s["rfilename"]: s.get("size") for s in sibs}


def gguf_variants(files):
    """One entry per variant, split shards grouped together."""
    groups = {}
    for f in files:
        if not f.lower().endswith(".gguf"):
            continue
        base, folder = os.path.basename(f), os.path.dirname(f)
        m = _SPLIT_RE.match(base)
        stem = m.group("stem") if m else base[:-5]
        groups.setdefault((folder, stem), []).append(f)
    return {k: sorted(v) for k, v in groups.items()}


def choose_variant(variants):
    def tag_of(stem):
        low = stem.lower()
        for t in BEST_TO_WORST:
            if t in low:
                return t
        return None

    scored = []
    for (folder, stem), files in variants.items():
        t = tag_of(stem)
        if t is not None:
            scored.append((BEST_TO_WORST.index(t), t, stem, files))
    if not scored:
        (folder, stem), files = sorted(variants.items())[0]
        print(f"[model] no quant tag in any filename; using {stem}")
        return stem, files
    scored.sort()

    if PRECISION == "full":
        for rank, t, stem, files in scored:
            if t in ("f32", "bf16", "f16", "fp16"):
                if NEEDS_GB[t] > VRAM_GB:
                    print(f"[warn] {t.upper()} wants ~{NEEDS_GB[t]} GB and this card "
                          f"has {VRAM_GB:.0f} GB. It will spill to system RAM and be "
                          "unusably slow. PRECISION = \"auto\" is the right setting "
                          "in a notebook.")
                print(f"[model] full precision → {stem} ({t.upper()})")
                return stem, files
        print(f"[warn] no F16/BF16 in this repo; taking the largest: {scored[0][2]}")
        return scored[0][2], scored[0][3]

    if PRECISION == "fast":
        # Speed is bandwidth / model-size, so the fastest usable variant is the
        # SMALLEST one whose quality still holds up — the opposite of what
        # "auto" does. Walk the preference order, not the size order.
        by_tag = {}
        for rank, t, stem, files in scored:
            by_tag.setdefault(t, (stem, files))
        for t in FAST_ORDER:
            if t not in by_tag:
                continue
            stem, files = by_tag[t]
            if NEEDS_GB[t] > VRAM_GB:
                continue
            gb = variant_gb(files)
            if gb and gb + MIN_FREE_GB > FREE_GB:
                print(f"[model]   skipping {stem}: {gb:.1f} GB on disk vs "
                      f"{FREE_GB:.1f} GB free")
                continue
            if gb:
                ceil = GPU_BANDWIDTH_GBS / gb

_RE_OFFLOAD = re.compile(r"offloaded\s+(\d+)\s*/\s*(\d+)\s+layers?\s+to\s+gpu")
_RE_USING = re.compile(r"using device (\S+)\s*\((.*?)\)", re.I)

_RE_CUDA = re.compile(r"Device\s+\d+:\s*(.+?),\s*compute capability")


def note_line(line):
    low = line.lower()
    m = _RE_OFFLOAD.search(low)
    if m:
        DEVICE_INFO["offloaded"] = int(m.group(1))
        DEVICE_INFO["layers"] = int(m.group(2))
    if not DEVICE_INFO["software"]:

                        (_RE_CUDA, "CUDA: {0}")):

            if mm and not DEVICE_INFO["device"]:
                DEVICE_INFO["device"] = fmt.format(*mm.groups())
                break

        DEVICE_INFO["software"] = True

# Performance flags. This card is an RTX PRO 6000 Blackwell class part; the
# defaults leave a lot on the table. Flash attention and a bigger micro-batch
# are the two that matter most for prefill. Not every llama.cpp build accepts
# every spelling, so if the server rejects them we retry with the plain set
# rather than refusing to run.
# Tiers, not one all-or-nothing list. An older llama.cpp that rejects a single
# flag used to drop us all the way to stock settings, losing the batch tuning
# too. Try the most aggressive set first and step down one tier at a time.
#
#   -fa              flash attention: large prefill win, and required for the
#                    quantised KV cache below.
#   -ctk/-ctv q8_0   quantise the KV cache. Halves the memory it reads per
#                    token, which is the other bandwidth cost after the weights,
#                    and matters more the longer the conversation gets.
#   -b / -ub         bigger batch and micro-batch: this is what actually feeds
#                    a card this wide during prefill. The default 512 leaves
#                    most of it idle.
#   --split-mode none / --main-gpu 0
#                    one GPU: skip the multi-GPU scheduling path entirely.
FLAG_TIERS = [
    ["-fa", "-ctk", "q8_0", "-ctv", "q8_0", "-b", "4096", "-ub", "2048",
     "--split-mode", "none", "--main-gpu", "0",
     "-t", str(os.cpu_count() or 8), "--no-warmup"],
    ["-fa", "-b", "4096", "-ub", "2048",
     "-t", str(os.cpu_count() or 8), "--no-warmup"],
    ["-b", "2048", "-ub", "1024", "-t", str(os.cpu_count() or 8)],
    [],
]

@app.post("/v1/chat/completions")
async def chat_completions(request: Request):
    check_auth(request)
    body = await request.json()
    if not body.get("messages"):
        raise HTTPException(status_code=400, detail="messages required")
    body["model"] = MODEL_ID
    if body.get("max_tokens"):
        body["max_tokens"] = min(int(body["max_tokens"]), MAX_NEW)
    stream = bool(body.get("stream", False))

    _bump(1)
    if not stream:
        try:
            r = await _client.post(LLAMA + "/v1/chat/completions", json=body)
            return JSONResponse(status_code=r.status_code, content=r.json())
        except Exception as e:
            return JSONResponse(status_code=500, content={"error": {
                "message": "%s: %s" % (type(e).__name__, e), "type": "generation_error"}})
        finally:
            _bump(-1)

    # A Cloudflare quick tunnel buffers a body unless told plainly not to, which
    # is what made replies arrive in one lump minutes later instead of
    # streaming. Needs all three: the headers below, bytes immediately, and
    # bytes every few seconds while the model prefills. ":" lines are SSE
    # comments the browser ignores.
    async def relay():
        q = asyncio.Queue()
        got_content = asyncio.Event()

        async def pump():
            try:
                async with _client.stream(
                        "POST", LLAMA + "/v1/chat/completions", json=body,
                        headers={"Accept": "text/event-stream",
                                 "Accept-Encoding": "identity"}) as r:
                    if r.status_code != 200:
                        detail = (await r.aread()).decode("utf-8", "replace")[:400]
                        await q.put(_err_chunk("\n[llama.cpp returned %s: %s]"
                                               % (r.status_code, detail)))
                        await q.put(b"data: [DONE]\n\n")
                        return
                    async for chunk in r.aiter_bytes():
                        got_content.set()
                        await q.put(chunk)
            except Exception as e:
                tail = "".join(_LOG_TAIL[-6:]).strip().replace("\n", " ")[:300]
                await q.put(_err_chunk(
                    "\n[Generation failed on the server: %s: %s. Last server log: %s]"
                    % (type(e).__name__, e, tail)))
                await q.put(b"data: [DONE]\n\n")
            finally:
                await q.put(None)

        task = asyncio.create_task(pump())
        yield b": open\n\n"
        try:
            while True:
                try:
                    item = await asyncio.wait_for(q.get(), timeout=5)
                except asyncio.TimeoutError:
                    if not got_content.is_set():
                        yield b": ping\n\n"
                    continue
                if item is None:
                    break
                yield item
        finally:
            task.cancel()
            _bump(-1)

    return StreamingResponse(relay(), media_type="text/event-stream", headers={
        "Cache-Control": "no-cache, no-store, no-transform",
        "X-Accel-Buffering": "no",
        "Connection": "keep-alive"})

Why this one is built differently. That repo ships GGUF, not safetensors, so the backend here is llama.cpp rather than transformers. The cell compiles llama-server with CUDA for your exact card the first time (5–15 minutes, then cached), asks the Hub which files the repo actually contains, downloads the right one, and puts a thin FastAPI layer in front so the site sees the identical /health and /v1/chat/completions contract as the normal server. Nothing in the chat needs to change.

Full precision means the F16 file. A GGUF repo ships one file per quantisation, so PRECISION = "full" pulls the F16/BF16 weights — roughly 54 GB, nothing quantised, every layer on the GPU. That wants about 64 GB of VRAM once the 32k KV cache is in there. On a smaller card use PRECISION = "auto" and it takes the largest file that fits, or name a quant directly with "q8_0" or "q4_k_m". Large files come as shards; the script fetches all of them and hands llama.cpp the first.

When it prints HERETIC API IS LIVE, paste the Public URL into settings the same way, then set the model field to corx3.8-heretic. Sandbox, search, files, GitHub and personalities all work exactly as they do on the normal build.

If the chat says it “did not respond within 25s”. That is almost never the address. The usual cause is the model running on the CPU while everything still looks fine: a Vulkan build on a notebook box picks up lavapipe, a software rasteriser, so -ngl 999 reports “63/63 layers offloaded to GPU” and then generates at a few seconds per token. A 27B is unusable there, and the front end can only report the silence.

This cell will not let that happen quietly. It reads the startup log for the device it actually got and refuses to hand you a URL if the model landed on the CPU. Before printing the address it runs a warm-up generation and tells you the measured tokens/sec — so you know it is fast enough without finding out in the chat. Set REQUIRE_GPU = False if you want it to serve anyway.

If it said “CUDA Toolkit not found”, that is the opposite problem and just as common: a notebook container ships the NVIDIA driver but no toolkit, so nvidia-smi works, nvcc does not exist, and there is nothing to compile with. No pip wheel fixes it either — nvidia-cuda-nvcc-cu12 contains ptxas and nvvm only, not nvcc.

So getting a backend is a ladder, and the cell walks it for you: compile with CUDA if nvcc is already there; else use a prebuilt CUDA binary if one is published; else use the real NVIDIA Vulkan driver if it is installed, which is a genuine GPU path needing no toolkit at all; else install about 1 GB of CUDA from NVIDIA’s own apt repo and compile. A software Vulkan device is refused at every rung. If none of it is possible it stops with a list of what to change — it will never quietly serve from the CPU. Pin a rung with BACKEND = "cuda" or "vulkan", and set INSTALL_CUDA_TOOLKIT = False to be told rather than fixed.

The chat helps from its side too: when your own server misses the window it asks /health — which answers instantly even while generation crawls — and tells you which problem you have, a dead tunnel URL or a model on the wrong device, quoting the device and tok/s it reported.

Optional · Voice studio

Clone a voice with TriStream-SVS.

A second server, for the microphone button in the chat header. Give it a voice to clone and a performance to sing, and it returns the performance in that voice. Runs alongside the chat model on the same machine, or on its own.

How the two clips are used. TriStream keeps pitch, timbre and texture in three encoder streams that cannot exchange information until the fusion trunk, so a two-clip transplant is the natural operation rather than a trick. The voice clip goes through the speaker encoder into the filter stream — the only stream that carries identity. The performance clip contributes its F0 contour to the source stream, which has no pathway to timbre at all, and its texture to the residual stream. That is why the melody transfers without the original singer’s voice coming with it.

The tunnel URL is public and keyless by default, exactly like the chat server. Anyone holding it can run voice cloning on your GPU. Clone voices you have the right to use — your own, or ones you have permission for.

def pids_on_port(port):
    """Who is listening on this port, using /proc alone.

    lsof and fuser are both absent from most notebook images, so relying on
    them means the port never actually gets cleared and the failure looks
    identical to not having tried. /proc/net/tcp gives the socket inode of
    every listener; /proc/<pid>/fd maps inodes back to processes.
    """
    inodes = set()
    for proto in ("tcp", "tcp6"):
        try:
            rows = open("/proc/net/" + proto).read().splitlines()[1:]
        except OSError:
            continue
        for row in rows:
            f = row.split()
            if len(f) < 10 or f[3] != "0A":        # 0A is TCP_LISTEN
                continue
            try:
                if int(f[1].rsplit(":", 1)[1], 16) == port:
                    inodes.add(f[9])
            except (IndexError, ValueError):
                continue
    if not inodes:
        return []

    found = []
    for entry in os.listdir("/proc"):
        if not entry.isdigit():
            continue
        fd_dir = "/proc/%s/fd" % entry
        try:
            names = os.listdir(fd_dir)
        except OSError:
            continue                                # gone, or not ours to read
        for fd in names:
            try:
                target = os.readlink(os.path.join(fd_dir, fd))
            except OSError:
                continue
            if target.startswith("socket:[") and target[8:-1] in inodes:
                found.append(int(entry))
                break
    return found

def stop_previous():
    """Stop the instance from a previous run, by pid file then by port."""
    if os.path.exists(PID_PATH):
        try:
            old = int(open(PID_PATH).read().strip())
        except Exception:
            old = 0
        try:
            if old > 0 and old != os.getpid() and is_our_runner(old):
                for sig in (signal.SIGTERM, signal.SIGKILL):
                    try:
                        # start_new_session made it a process group leader, so
                        # this takes its cloudflared child with it.
                        os.killpg(os.getpgid(old), sig)
                        print("stopped the previous server (pid %d)" % old)
                    except ProcessLookupError:
                        break
                    except Exception:

                            os.kill(old, sig)
                        except Exception:
                            break
                    freed = False
                    for _ in range(20):
                        if port_free(PORT):
                            freed = True
                            break
                        time.sleep(0.25)
                    if freed:
                        return
        finally:
            # Always, including on the early return above: a pid file left
            # behind points at a process that no longer exists, and eventually
            # at an unrelated one that does.
            try:
                os.remove(PID_PATH)
            except OSError:
                pass

    # if it is one of ours.

def load_audio(raw, target_sr=None):
    """Bytes -> mono float32 at target_sr."""
    target_sr = target_sr or SAMPLE_RATE
    import librosa
    wav, sr = sf.read(io.BytesIO(raw), dtype="float32", always_2d=True)

def f0_contour(wav, sr=None, fmin=65.0, fmax=1200.0):
    """Log-F0 and the voiced/unvoiced flag — exactly what the source stream takes.

    The source stream receives these two arrays and nothing else. It has no path
    to the mel, the singer embedding, or any other timbre-carrying signal, which
    is why taking them from a different singer transfers the melody without
    dragging that singer's voice along with it.
    """
    import librosa
    sr = sr or SAMPLE_RATE
    f0, voiced, _ = librosa.pyin(wav, fmin=fmin, fmax=fmax, sr=sr,
                                 frame_length=1024, hop_length=256)
    f0 = np.nan_to_num(f0, nan=0.0)
    log_f0 = np.zeros_like(f0)
    nz = f0 > 0
    log_f0[nz] = np.log(f0[nz])
    return log_f0.astype("float32"), voiced.astype("float32")


def mel_spec(wav, sr=None, n_mels=100):
    """100-band mel at 24 kHz — the decoder's output format, per the model card."""
    import librosa
    sr = sr or SAMPLE_RATE
    m = librosa.feature.melspectrogram(y=wav, sr=sr, n_fft=1024, hop_length=256,
                                       n_mels=n_mels)
    return np.log(np.clip(m, 1e-5, None)).astype("float32")


def shift_pitch(log_f0, semitones):
    if not semitones:
        return log_f0
    out = log_f0.copy()
    nz = out > 0
    out[nz] = out[nz] + float(semitones) * np.log(2.0) / 12.0
    return out

def walk_repo(root):
    """Every file in the snapshot, relative to it.

    listdir() was the original bug: it sees only the top level, so a repo that
    keeps its code in a subdirectory looks like a repo with no code at all.
    """
    out = []

        dirnames[:] = [d for d in dirnames if d not in (".git", ".cache", "__pycache__")]
        for f in filenames:
            out.append(os.path.relpath(os.path.join(dirpath, f), root))
    return sorted(out)


def add_import_roots(root, files):
    """Put every directory that holds Python on sys.path.

    This matters twice over. It lets a module inside a subdirectory be imported
    at all, and it lets pickle resolve the module path recorded inside a saved
    nn.Module — unpickling fails with ModuleNotFoundError unless the defining
    module is importable under the exact name it had when it was saved.
    """
    roots = {root}
    for rel in files:
        if rel.endswith(".py"):
            d = os.path.dirname(os.path.join(root, rel))
            roots.add(d)
            # A package's parent is what makes "pkg.module" resolve.
            if os.path.exists(os.path.join(d, "__init__.py")):
                roots.add(os.path.dirname(d))
    for d in sorted(roots, key=len):
        if d and d not in sys.path:
            sys.path.insert(0, d)
    return sorted(roots)


# Files that are programs rather than model definitions. Importing one runs it:
# a training script parses argv and calls sys.exit(), a setup.py builds a
# package, a test file may assert its way out. None of them define the model.
SKIP_PY = ("setup.py", "__main__.py", "conftest.py", "train.py", "training.py",
           "preprocess.py", "app.py", "demo.py", "gradio_app.py", "webui.py")
SKIP_DIRS = ("tests", "test", "scripts", "examples", "notebooks", "data")


def worth_importing(rel):
    base = os.path.basename(rel)
    if base in SKIP_PY or base.startswith("test_"):
        return False
    parts = os.path.dirname(rel).split(os.sep)
    return not any(p in SKIP_DIRS for p in parts)


def import_file(path, root):
    """Import one .py by location, without guessing a package name.

    Running the repo's Python is the same trust you extend by running its
    weights; there is no way to load a model like this without it. What is not
    acceptable is letting it end the process — argv is blanked so a stray
    argparse cannot exit, and SystemExit is turned back into a plain error.
    """
    import importlib.util
    name = os.path.splitext(os.path.relpath(path, root))[0].replace(os.sep, "_")
    spec = importlib.util.spec_from_file_location(name, path)
    if spec is None or spec.loader is None:
        raise ImportError("no loader for %s" % path)
    mod = importlib.util.module_from_spec(spec)
    saved_argv = sys.argv
    sys.argv = [os.path.basename(path)]
    sys.modules[name] = mod
    try:
        spec.loader.exec_module(mod)
    except SystemExit as e:
        sys.modules.pop(name, None)
        raise ImportError("the file called sys.exit(%s) on import — it is a "
                          "script, not a model definition" % e.code)
    except BaseException:
        sys.modules.pop(name, None)
        raise
    finally:
        sys.argv = saved_argv
    return mod


def load_checkpoint(path):
    """torch.load with the trust decision made explicitly.

    torch 2.6 flipped weights_only to True, which refuses any checkpoint holding
    more than tensors. These files predate that and may hold a pickled module,
    so the flag has to come off deliberately rather than by accident.
    """
    try:
        return torch.load(path, map_location="cpu", weights_only=False)
    except TypeError:
        return torch.load(path, map_location="cpu")


def find_state_dict(blob):
    """Pull the weights out of whatever shape the checkpoint was saved in."""
    if hasattr(blob, "state_dict") and callable(getattr(blob, "state_dict")):
        return None, "(already a module)"
    if not isinstance(blob, dict):
        return None, None

    def looks_like_weights(d):
        vals = [v for v in list(d.values())[:12]]
        return bool(vals) and all(isinstance(v, torch.Tensor) for v in vals)

    for key in ("state_dict", "model_state_dict", "model", "generator", "net",
                "module", "weights", "params", "g"):
        v = blob.get(key)
        if isinstance(v, dict) and looks_like_weights(v):
            return v, key
        if hasattr(v, "state_dict"):
            return v, key
    if looks_like_weights(blob):
        return blob, "(top level)"
    return None, None


def strip_prefix(sd):
    """DataParallel and Lightning both leave a prefix on every key."""
    for pre in ("module.", "model.", "_orig_mod."):
        if sd and all(k.startswith(pre) for k in sd):
            return {k[len(pre):]: v for k, v in sd.items()}
    return sd


def candidate_classes(mods):
    """nn.Module subclasses defined in the repo's own modules, best guess first."""
    found = []
    for mod in mods:
        for attr in dir(mod):
            obj = getattr(mod, attr, None)
            if not isinstance(obj, type) or not issubclass(obj, torch.nn.Module):
                continue
            if obj.__module__ != mod.__name__:      # imported from torch, not defined here
                continue
            score = sum(2 for h in CLASS_HINTS if h in attr.lower())
            found.append((score, attr, obj))
    found.sort(key=lambda t: -t[0])
    return found


def construct(cls, cfg):
    """Try the constructor shapes a config-driven model usually takes."""
    for args, kwargs in ((( ), cfg if isinstance(cfg, dict) else {}),
                         ((cfg,), {}),
                         (( ), {})):
        try:
            return cls(*args, **kwargs), ("%s(**config)" % cls.__name__ if kwargs
                                          else "%s(config)" % cls.__name__ if args
                                          else "%s()" % cls.__name__)
        except Exception:
            continue
    return None, None


def load_model_def():
    """Import a model definition supplied from outside the repo.

    This repo publishes parameters without a network, so the class has to come
    from wherever the model was trained. Accepts a local path or an https URL,
    and falls back to <workdir>/model_def.py so a file can simply be dropped in.
    Whatever it defines is still checked against the checkpoint tensor by tensor
    before anything uses it — supplying a definition is not a promise that it is
    the right one.
    """
    src = MODEL_DEF or os.path.join(WORKDIR, "model_def.py")
    if src.startswith(("http://", "https://")):
        dest = os.path.join(WORKDIR, "model_def.py")
        try:
            import urllib.request
            urllib.request.urlretrieve(src, dest)
            log("fetched the model definition from", src)
            src = dest
        except Exception as e:
            STATE["tried"].append("model definition %s: %s" % (src, str(e)[:140]))
            return []
    if not os.path.exists(src):
        return []
    try:
        mod = import_file(src, os.path.dirname(src) or ".")
    except KeyboardInterrupt:
        raise
    except BaseException as e:
        STATE["tried"].append("model definition %s: %s: %s"
                              % (src, type(e).__name__, str(e)[:160]))
        return []
    names = [n for _, n, _ in candidate_classes([mod])]
    log("model definition %s defines: %s" % (src, ", ".join(names) or "no nn.Module subclass"))
    STATE["tried"].append("model definition %s imported (classes: %s)"
                          % (src, ", ".join(names) or "none"))
    return [mod]


def describe_arch(sd, cfg, blob, rel):
    """Write down everything needed to rebuild the missing model class.

    A state_dict is a complete description of a network's parameter tree: every
    submodule path and every tensor shape. Together with the config and the
    checkpoint's own metadata that is enough to write a matching nn.Module —
    which is the only way forward for a repo that ships weights without code.
    Guessing the architecture instead would load some tensors, skip others, and
    synthesise noise while reporting success.
    """
    groups = {}
    for k, v in sd.items():
        top = k.split(".")[0]
        g = groups.setdefault(top, {"count": 0, "params": 0, "sample": []})
        g["count"] += 1
        try:
            g["params"] += int(v.numel())
        except Exception:
            pass
        if len(g["sample"]) < 6:
            g["sample"].append("%s %s" % (k, tuple(getattr(v, "shape", ()))))

    meta = {}
    if isinstance(blob, dict):
        for k, v in blob.items():
            if k == "model" or isinstance(v, dict) and len(v) > 50:
                continue
            if isinstance(v, (str, int, float, bool)) or v is None:
                meta[k] = v
            elif isinstance(v, dict):
                meta[k] = {kk: vv for kk, vv in list(v.items())[:40]
                           if isinstance(vv, (str, int, float, bool)) or vv is None}

    STATE["arch"] = {
        "checkpoint": rel,
        "tensors": len(sd),
        "parameters": sum(g["params"] for g in groups.values()),
        "config": cfg,
        "checkpoint_meta": meta,
        "top_level_modules": {k: {"tensors": g["count"], "parameters": g["params"],
                                  "sample": g["sample"]}
                              for k, g in sorted(groups.items())},
        "all_keys": ["%s %s" % (k, tuple(getattr(v, "shape", ()))) for k, v in sd.items()],
    }

    path = os.path.join(WORKDIR, "architecture.json")
    try:
        with open(path, "w") as fh:
            json.dump(STATE["arch"], fh, indent=2, default=str)
    except Exception as e:
        log("could not write %s: %s" % (path, e))

    log("")
    log("=" * 72)
    log("  This repo has weights but no model code.")
    log("  %d tensors, %s parameters, %d top-level modules:"
        % (len(sd), format(STATE["arch"]["parameters"], ","), len(groups)))
    for name, g in sorted(groups.items(), key=lambda kv: -kv[1]["params"]):
        log("     %-24s %4d tensors  %s params" % (name, g["count"], format(g["params"], ",")))
    log("")
    log("  The full parameter tree is in %s" % path)
    log("  and at <tunnel-url>/architecture. That is what a matching nn.Module")
    log("  has to be written against — the shapes name every layer.")
    log("=" * 72)
    log("")


def bind_model():
    from huggingface_hub import snapshot_download
    local = snapshot_download(repo_id=REPO_ID, local_dir=os.path.join(WORKDIR, "model"))

    files = walk_repo(local)
    STATE["files"] = files
    log("repo files (%d):" % len(files))
    for f in files[:60]:
        log("   ", f)

    roots = add_import_roots(local, files)
    log("import roots:", roots)

    cfg = {}
    cfg_path = os.path.join(local, "config.json")
    if os.path.exists(cfg_path):
        try:
            with open(cfg_path) as fh:
                cfg = json.load(fh)
            log("config keys:", sorted(cfg)[:30])
        except Exception as e:
            STATE["tried"].append("config.json: %s" % str(e)[:120])

    obj = None

    # (a) torch.hub, if the repo ships a hubconf anywhere in the tree
    for rel in files:
        if os.path.basename(rel) != "hubconf.py":
            continue
        hub_dir = os.path.dirname(os.path.join(local, rel)) or local
        try:
            hub = import_file(os.path.join(local, rel), local)
            entries = [a for a in dir(hub)
                       if not a.startswith("_") and callable(getattr(hub, a))]
            log("hubconf entry points:", entries)
            for entry in entries:
                try:
                    obj = torch.hub.load(hub_dir, entry, source="local", pretrained=True)
                    STATE["detail"] = "loaded via torch.hub %s()" % entry
                    break
                except Exception as e:
                    STATE["tried"].append("hub.%s: %s" % (entry, str(e)[:120]))
        except Exception as e:
            STATE["tried"].append("hubconf: %s" % str(e)[:140])
        if obj is not None:
            break

    # (b) transformers — only when the config is actually a transformers config.
    #     Trying it without a model_type key just buries the real error under a
    #     guaranteed failure, which is exactly what happened the first time.
    if obj is None and cfg.get("model_type"):
        for loader in ("AutoModel", "AutoModelForSpeechSeq2Seq"):
            try:
                import transformers
                cls = getattr(transformers, loader)
                obj = cls.from_pretrained(local, trust_remote_code=True,
                                          torch_dtype=DTYPE).to(DEVICE).eval()
                STATE["detail"] = "loaded via transformers.%s(trust_remote_code=True)" % loader
                break
            except Exception as e:
                STATE["tried"].append("%s: %s" % (loader, str(e)[:160]))
    elif obj is None and os.path.exists(cfg_path):
        STATE["tried"].append("transformers: skipped, config.json has no model_type "
                              "(it is the model's own config, not a transformers one)")

    # (c) a module that builds the model — supplied by the user first, since a
    #     definition someone went and fetched beats anything guessed from the
    #     repo, then anything the repo itself ships.
    repo_mods = load_model_def()
    if obj is None:
        for rel in files:
            if not rel.endswith(".py") or os.path.basename(rel) == "hubconf.py":
                continue
            if not worth_importing(rel):
                STATE["tried"].append("skipped %s (a script, not a model definition)" % rel)
                continue
            try:
                repo_mods.append(import_file(os.path.join(local, rel), local))
            except KeyboardInterrupt:
                raise
            except BaseException as e:
                STATE["tried"].append("import %s: %s: %s"
                                      % (rel, type(e).__name__, str(e)[:120]))
        log("imported %d repo module(s)" % len(repo_mods))

        for mod in repo_mods:
            for factory in ("load_model", "load_pretrained", "from_pretrained",
                            "build_model", "build", "get_model", "load"):
                fn = getattr(mod, factory, None)
                if fn is None:
                    continue
                # A bare class constructor is not a loader. Calling TriStreamSVS()
                # succeeds and hands back a randomly initialised network, which
                # would then report ready and synthesise noise — far worse than
                # failing. Classes are only tried with the repo path, which at
                # least implies they read something; weights that come in through
                # a checkpoint are checked tensor by tensor further down.
                arg_sets = ((local,),) if isinstance(fn, type) else ((local,), ())
                for args in arg_sets:
                    try:
                        made = fn(*args)
                    except Exception as e:
                        STATE["tried"].append("%s.%s: %s"
                                              % (mod.__name__, factory, str(e)[:120]))
                        continue
                    if made is not None:
                        obj = made
                        STATE["detail"] = "loaded via %s.%s()" % (mod.__name__, factory)
                        break
                if obj is not None:
                    break
            if obj is not None:
                break

    # (d) the checkpoint itself. For a weights-first repo this is the real path,
    #     and the original script never opened these files at all.
    ckpts = [f for f in files if f.lower().endswith(CKPT_EXT)]
    ckpts.sort(key=lambda f: next((i for i, k in enumerate(CKPT_RANK)
                                   if k in os.path.basename(f).lower()), len(CKPT_RANK)))
    if obj is None and ckpts:
        for rel in ckpts:
            path = os.path.join(local, rel)
            try:
                blob = load_checkpoint(path)
            except ModuleNotFoundError as e:
                # The single most useful diagnostic there is: the pickle names
                # the exact module the repo expects to be importable.
                STATE["tried"].append(
                    "%s: needs the module '%s', which is not in this repo — the "
                    "architecture code lives somewhere else" % (rel, e.name))
                continue
            except Exception as e:
                STATE["tried"].append("%s: %s" % (rel, str(e)[:160]))
                continue

            if isinstance(blob, torch.nn.Module):
                obj = blob
                STATE["detail"] = "loaded the pickled module out of %s" % rel
                break

            sd, where = find_state_dict(blob)
            if isinstance(blob, dict):
                log("%s top-level keys: %s" % (rel, sorted(blob)[:20]))
                # Checkpoints often carry the config the model was built with.
                for ck in ("config", "hyper_parameters", "hparams", "args", "cfg"):
                    if isinstance(blob.get(ck), dict) and not cfg:
                        cfg = blob[ck]
                        log("using config from the checkpoint's '%s' key" % ck)
            if sd is None:
                STATE["tried"].append("%s: no state_dict inside (keys: %s)"
                                      % (rel, sorted(blob)[:12] if isinstance(blob, dict)
                                         else type(blob).__name__))
                continue
            if hasattr(sd, "state_dict"):
                obj = sd
                STATE["detail"] = "loaded the pickled module from %s['%s']" % (rel, where)
                break

            sd = strip_prefix(sd)
            STATE["ckpt_keys"] = list(sd)[:12]
            log("%s: %d weight tensors under '%s'" % (rel, len(sd), where))

            classes = candidate_classes(repo_mods)
            if not classes:
                # The weights are here and readable; the class to put them in is
                # not. That is recoverable — a state_dict names every submodule
                # and every shape — so write down the blueprint rather than only
                # the complaint. Nothing can be reconstructed from a sentence
                # saying it failed.
                # Only the first, which is the best-ranked checkpoint — letting
                # the later one overwrite it would describe the worse weights.
                if STATE["arch"] is None:
                    describe_arch(sd, cfg, blob, rel)
                STATE["tried"].append(
                    "%s: %d weight tensors, but this repo ships no architecture "
                    "code to load them into — see /architecture" % (rel, len(sd)))
                continue
            for _, name, cls in classes[:6]:
                made, how = construct(cls, cfg)
                if made is None:
                    STATE["tried"].append("%s(): could not construct from config" % name)
                    continue
                try:
                    missing, unexpected = made.load_state_dict(sd, strict=False)
                except Exception as e:
                    STATE["tried"].append("%s.load_state_dict: %s" % (name, str(e)[:120]))
                    continue
                matched = len(sd) - len(unexpected)
                # A class that accepts a handful of keys is the wrong class; a
                # real match takes nearly all of them.
                if matched >= max(8, int(0.6 * len(sd))):
                    obj = made
                    STATE["detail"] = ("built %s from config and loaded %d/%d tensors "
                                       "from %s (%d missing)"
                                       % (how, matched, len(sd), rel, len(missing)))
                    break
                STATE["tried"].append("%s: only %d/%d tensors matched"
                                      % (name, matched, len(sd)))
            if obj is not None:
                break

    if obj is None:
        STATE["ready"] = False
        STATE["detail"] = summarise_failure(files, ckpts)
        log("!! " + STATE["detail"])
        return

    try:
        obj = obj.to(DEVICE).eval()
    except Exception as e:
        STATE["tried"].append("to(%s): %s" % (DEVICE, str(e)[:120]))

    # Which call do we have? Stream-level beats whole-clip conversion.
    for name in STREAM_CALLS:
        if callable(getattr(obj, name, None)):
            STATE["binding"] = ("streams", name)
            break
    else:
        for name in CONVERT_CALLS:
            if callable(getattr(obj, name, None)):
                STATE["binding"] = ("convert", name)
                break

    if STATE["binding"] is None:
        STATE["ready"] = False
        methods = [m for m in dir(obj) if not m.startswith("_")
                   and callable(getattr(obj, m, None))][:40]
        STATE["detail"] = ("Built %s, but none of its methods are a recognised "
                           "inference call. Public methods: %s"
                           % (type(obj).__name__, ", ".join(methods)))
        log("!! " + STATE["detail"])
        return

    STATE["model"] = obj
    STATE["ready"] = True
    log("bound to %s.%s — %s" % (type(obj).__name__, STATE["binding"][1], STATE["detail"]))


def summarise_failure(files, ckpts):
    """Say what is actually in the repo, so the next step is obvious.

    A failure here is nearly always one of two things: the architecture code is
    not in the repo, or it is there under a shape this script did not try. The
    difference is visible in the file list, so print it rather than a verdict.
    """
    py = [f for f in files if f.endswith(".py")]
    lines = ["Could not construct the model from this repo."]
    if not py and STATE["arch"]:
        a = STATE["arch"]
        lines.append(
            "The weights read fine — %d tensors, %s parameters, in %d top-level "
            "modules (%s) — but the repo ships no Python, so there is no class to "
            "load them into. A checkpoint cannot be run on its own: it is the "
            "parameters, not the network. The full parameter tree is at "
            "/architecture and in the log, which is what a matching model "
            "definition has to be written against. If you have the training "
            "script this came from, that script already contains the class — "
            "set MODEL_DEF at the top of this file to its path or URL, or drop "
            "it at %s, and it will be constructed from config.json and checked "
            "against the checkpoint."
            % (a["tensors"], format(a["parameters"], ","),
               len(a["top_level_modules"]), ", ".join(list(a["top_level_modules"])[:8]),
               os.path.join(WORKDIR, "model_def.py")))
        meta = a.get("checkpoint_meta") or {}
        if meta.get("developer") or meta.get("name"):
            lines.append("The checkpoint says it was trained by %s as \"%s\"%s."
                         % (meta.get("developer", "someone"),
                            meta.get("name", "this model"),
                            ", step %s" % meta["step"] if meta.get("step") else ""))
    elif not py:
        lines.append(
            "It contains weights (%s) but no Python at all, so there is no "
            "architecture to load them into."
            % (", ".join(os.path.basename(c) for c in ckpts) or "none found"))
    else:
        lines.append("Python found: %s." % ", ".join(py))
    lines.append("Files: %s." % ", ".join(files[:25]))
    if STATE["ckpt_keys"]:
        lines.append("First weight names: %s." % ", ".join(STATE["ckpt_keys"]))
    lines.append("Attempts: %s" % (" | ".join(STATE["tried"]) or "none"))
    return " ".join(lines)

@torch.inference_mode()
def run_convert(voice_wav, perf_wav, residual_from="performance", pitch_shift=0.0,
                steps=STEPS, lyrics=""):
    """Sing the performance clip in the voice clip's voice."""
    model = STATE["model"]
    kind, name = STATE["binding"]
    fn = getattr(model, name)

    log_f0, vuv = f0_contour(perf_wav)
    log_f0 = shift_pitch(log_f0, pitch_shift)
    residual_src = perf_wav if residual_from == "performance" else voice_wav

    if kind == "streams":
        # The real transplant: each stream driven from the clip it belongs to.
        out = fn(
            source={"log_f0": log_f0, "voiced": vuv},
            filter={"reference_audio": voice_wav, "sample_rate": SAMPLE_RATE},
            residual={"audio": residual_src, "sample_rate": SAMPLE_RATE},
            lyrics=lyrics or None,
            steps=int(steps),
        )
        used = "stream transplant (%s)" % name
    else:
        # Repo exposes only whole-clip conversion. Still the right mapping —
        # content from the performance, identity from the voice — but the stream
        # routing is the repo's, not ours.
        out = _call_flexible(fn, perf_wav, voice_wav, steps, lyrics)
        used = "repo conversion call (%s)" % name

    return _as_audio(out), used


@torch.inference_mode()
def run_generate(voice_wav, lyrics, melody_wav=None, steps=STEPS, pitch_shift=0.0):
    """Generation mode: sing lyrics from scratch in the cloned voice."""
    model = STATE["model"]
    kind, name = STATE["binding"]
    fn = getattr(model, name)

    kwargs = {"lyrics": lyrics, "steps": int(steps)}
    if melody_wav is not None:
        log_f0, vuv = f0_contour(melody_wav)
        log_f0 = shift_pitch(log_f0, pitch_shift)
        kwargs["source"] = {"log_f0": log_f0, "voiced": vuv}
    if kind == "streams":
        kwargs["filter"] = {"reference_audio": voice_wav, "sample_rate": SAMPLE_RATE}
        out = fn(**kwargs)
    else:
        out = _call_flexible(fn, None, voice_wav, steps, lyrics, melody_wav)
    return _as_audio(out), "generation (%s)" % name


def _call_flexible(fn, content_wav, reference_wav, steps, lyrics, melody_wav=None):
    """Try the argument spellings a repo of this kind plausibly uses.

    Every attempt and its error is kept, so a failure reports what was actually
    tried rather than a bare traceback.
    """
    import inspect
    attempts = []
    base = {"sample_rate": SAMPLE_RATE, "sr": SAMPLE_RATE, "steps": int(steps),
            "num_steps": int(steps), "n_steps": int(steps)}
    names_content = ["source_audio", "content_audio", "audio", "wav", "src", "source"]
    names_ref     = ["reference_audio", "speaker_audio", "target_audio", "ref", "reference",
                     "prompt_audio", "speaker_wav"]
    try:
        sig = set(inspect.signature(fn).parameters)
    except (TypeError, ValueError):
        sig = set()

    kwargs = {}
    if content_wav is not None:
        for n in names_content:
            if n in sig:
                kwargs[n] = content_wav
                break
    for n in names_ref:
        if n in sig:
            kwargs[n] = reference_wav
            break
    if lyrics and "lyrics" in sig:
        kwargs["lyrics"] = lyrics
    if melody_wav is not None and "melody" in sig:
        kwargs["melody"] = melody_wav
    for k, v in base.items():
        if k in sig:
            kwargs[k] = v

    if kwargs:
        try:
            return fn(**kwargs)
        except Exception as e:
            attempts.append("kwargs %s: %s" % (sorted(kwargs), str(e)[:200]))

    for args in ([content_wav, reference_wav] if content_wav is not None else [reference_wav],
                 [reference_wav, content_wav] if content_wav is not None else None):
        if args is None:
            continue
        try:
            return fn(*[a for a in args if a is not None])
        except Exception as e:
            attempts.append("positional %d: %s" % (len(args), str(e)[:200]))

    raise RuntimeError("No call signature worked. Tried: " + " | ".join(attempts))


def _as_audio(out):
    """Whatever came back -> a mono float32 waveform."""
    if isinstance(out, dict):
        for k in ("audio", "wav", "waveform", "samples", "output"):
            if k in out:
                out = out[k]
                break
    if isinstance(out, (list, tuple)) and out:
        out = out[0]
    if torch.is_tensor(out):
        out = out.detach().float().cpu().numpy()
    out = np.asarray(out, dtype="float32")
    if out.ndim > 1:
        out = out.reshape(out.shape[0], -1).mean(axis=0) if out.shape[0] < out.shape[-1] \
              else out.mean(axis=-1)
    peak = float(np.max(np.abs(out))) if out.size else 0.0
    if peak > 1.0:
        out = out / peak * 0.97
    return out

def load_in_background():
    """Load the model without the API depending on it finishing, or succeeding.

    This used to run inline, before uvicorn.run(). Loading a model means
    importing the repo's own Python, and a repo script that calls sys.exit()
    — argparse does exactly that when a training script is missing an argument
    — raises SystemExit, which is a BaseException and walks straight through
    `except Exception`. The process died, the tunnel stayed up with nothing
    behind it, and the browser got a Cloudflare 502 carrying no CORS headers,
    which surfaces as a bare "Failed to fetch" with nothing to diagnose.

    So: serve first, load second, and catch everything. A failed or slow load
    is now something /health can describe instead of a server that vanished.
    """
    try:
        bind_model()
    except KeyboardInterrupt:
        raise
    except BaseException as e:
        STATE["ready"] = False
        STATE["detail"] = "Load failed: %s: %s" % (type(e).__name__, e)
        traceback.print_exc()
    finally:
        STATE["loading"] = False
        if STATE["ready"]:
            log("model ready")
        else:
            log("!! model not ready — /health has the detail")


threading.Thread(target=load_in_background, daemon=True).start()
